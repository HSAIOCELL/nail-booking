function sendBooking() { const mainColorVal = document.getElementById('mainColor').value; const isMainColorSelected = mainColorVal !== ""; if (showJumpColor) { const jumpColorVal = document.getElementById('jumpColor').value; if (jumpColorVal === "") { alert("🎨 請選擇跳色色系，或關閉該項目！"); return; } } // 收集所有預約資料 const bookingDate = document.getElementById('bookingDate').value; const bookingHour = document.getElementById('bookingHour').value; const bookingMinute = document.getElementById('bookingMinute').value; const timeStr = `${bookingHour}:${bookingMinute}`; const totalPrice = document.getElementById('totalPrice').innerText; // 收集項目細節 let items = []; if (isMainColorSelected) items.push(`單色款式: ${mainColorVal}`); if (showJumpColor) items.push(`跳色: ${document.getElementById('jumpColor').value}`); if (showStyle) { const currentType = document.querySelector('input[name="styleType"]:checked'); if (currentType) { if (currentType.value === 'partial') { let styles = []; document.querySelectorAll('input[name="style"]:checked').forEach(cb => styles.push(cb.value)); items.push(`局部造型(${styles.join(',')})`); } else { let fullStyles = []; document.querySelectorAll('input[name="fullStyle"]:checked').forEach(cb => fullStyles.push(cb.value)); items.push(`十指造型(${fullStyles.join(',')})`); } } } if (needRemoval) items.push("需要卸甲"); // 準備要送往 Google 試算表的資料 const bookingData = { bookingId: 'M' + Date.now().toString().slice(-6), customerName: "LINE 貴賓", // 之後串接 LIFF 後會自動帶入真實 LINE 名字 date: bookingDate, time: timeStr, items: items.join(' / '), totalPrice: parseInt(totalPrice, 10) }; // 發送至 Google 試算表後端 const scriptUrl = "https://script.google.com/macros/s/AKfycbzu25rQUBTULTCgpqFnSfx21oVWyb1LoFnqQtGLGGUhJ6UFCHzu7po873wKRphQEQEkIg/exec"; // 顯示送出中提示 const submitBtn = document.querySelector('.submit-btn'); submitBtn.innerText = "預約送出中..."; submitBtn.disabled = true; fetch(scriptUrl, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(bookingData) }) .then(() => { alert("預約成功！資料已自動存入系統。"); // 同時引導開啟 LINE 官方帳號進行確認 const lineMessage = encodeURIComponent(`您好，我剛剛送出了預約：\n日期：${bookingDate} ${timeStr}\n金額：NT$ ${totalPrice}`); window.open(`https://line.me/R/oaMessage/${LINE_OFFICIAL_ACCOUNT_ID}/?${lineMessage}`, '_blank'); location.reload(); }) .catch(error => { console.error('Error:', error); alert("送出失敗，請再試一次或直接聯繫官方 LINE。"); submitBtn.innerText = "送出預約單 (開啟 LINE)"; submitBtn.disabled = false; }); }
